%%% Instructor / student and solutions toggles.
%%%   instructor           on: full title slide with reminders and poll code,
%%%                            poll frames shown.
%%%                        off: student version (title-only first slide, no polls).
%%%   instructorsolutions  on: worked solutions shown (solution frames replace
%%%                            prompt frames; instructorsol content visible).
%%% Leave both lines at the top of the document body as \toggletrue/\togglefalse
%%% so build_variants.py can rewrite them.
\usepackage{etoolbox}   %%% provides \newtoggle
\usepackage{environ}    %%% provides \NewEnviron
\newtoggle{instructor}
\newtoggle{instructorsolutions}
\newcommand{\ifinstructor}[2]{\iftoggle{instructor}{#1}{#2}}
%%%-------------------------------------------------------------------------
%%% Live polling. \pollcode is the session code students type in; \pollsite
%%% is where they go. Update \pollcode each class meeting (one place per deck).
\newcommand{\pollcode}{XYZXYZ}               %%% placeholder code
\newcommand{\pollsite}{polling.example.edu}  %%% COURSE: your polling site
\newcommand{\polltitle}{Poll question}       %%% COURSE: e.g. Quick check
%%% pollframe: a frame shown only in instructor builds, with the poll code in
%%% large type at the bottom right.
\NewEnviron{pollframe}{%
  \iftoggle{instructor}
    {\begin{frame}[vertical-alignment=top]
        \frametitle{\polltitle}
        \medskip
        \begin{minipage}[t][\textheight][s]{\linewidth}
            %%% wrapped in a minipage because ltx-talk doesn't
            %%% handle \vfill in an obvious way
            \BODY
        \vfill

{\raggedleft
 \fontsize{60pt}{60pt}\selectfont
 \pollcode\par}
\end{minipage}
    \end{frame}}
    {}%
  }
%%% instructorsol: content shown only when instructorsolutions is on; its
%%% space is kept (but nothing is drawn) when off.
\NewEnviron{instructorsol}{%
  \iftoggle{instructorsolutions}{%
    \BODY
  }{%
    \phantom{\BODY}
  }
}
